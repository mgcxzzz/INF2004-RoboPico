cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w CACHE STRING "Target board")
if(NOT DEFINED PICO_SDK_PATH)
    set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(buddy23_interface_v1 C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(buddy23_interface_v1 main.c motion_api.c nav_coordinator.c motion_executor.c nav_codec.c buddy_hooks.c)
target_link_libraries(buddy23_interface_v1 pico_stdlib hardware_adc hardware_pwm hardware_clocks hardware_sync hardware_flash m)
target_compile_options(buddy23_interface_v1 PRIVATE -Wall -Wextra -Werror)
pico_enable_stdio_uart(buddy23_interface_v1 0)
pico_enable_stdio_usb(buddy23_interface_v1 1)
target_compile_definitions(buddy23_interface_v1 PRIVATE
    LINE_CAL_MIN_SPAN=50
    CIRCUIT_SWEEP_SEARCH=1
    PICO_STDIO_USB_STDOUT_TIMEOUT_US=5000000 PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0)
option(INTERFACE_UNPACKED_REFERENCE "Measurement-only uncompressed interface state" OFF)
if(INTERFACE_UNPACKED_REFERENCE)
    target_compile_definitions(buddy23_interface_v1 PRIVATE INTERFACE_UNPACKED_REFERENCE=1)
endif()
set(BUDDY_EXTENSION_SOURCE "" CACHE FILEPATH "Optional Buddy5 extension with strong buddy_extension hooks")
if(BUDDY_EXTENSION_SOURCE)
    target_sources(buddy23_interface_v1 PRIVATE "${BUDDY_EXTENSION_SOURCE}")
endif()
pico_add_extra_outputs(buddy23_interface_v1)
