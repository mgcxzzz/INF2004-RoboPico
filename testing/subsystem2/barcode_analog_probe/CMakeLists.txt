cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w CACHE STRING "Target board")
if(NOT DEFINED PICO_SDK_PATH)
    set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(buddy23_barcode_analog_probe_v5 C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(buddy23_barcode_analog_probe_v5 main.c)
target_link_libraries(buddy23_barcode_analog_probe_v5 pico_stdlib hardware_adc hardware_flash hardware_sync m)
target_compile_options(buddy23_barcode_analog_probe_v5 PRIVATE -Wall -Wextra -Werror)
pico_enable_stdio_uart(buddy23_barcode_analog_probe_v5 0)
pico_enable_stdio_usb(buddy23_barcode_analog_probe_v5 1)
target_compile_definitions(buddy23_barcode_analog_probe_v5 PRIVATE
    PICO_STDIO_USB_STDOUT_TIMEOUT_US=5000000 PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0)
pico_add_extra_outputs(buddy23_barcode_analog_probe_v5)
